    #!/usr/bin/env python3
    """
    Revolve - AI Conversation Sync Script
    Syncs conversations from multiple AI providers to Obsidian vault.
    """

    import argparse
    import re
    import shutil
    from abc import ABC, abstractmethod
    from dataclasses import dataclass, field
    from datetime import datetime
    from pathlib import Path

    # --- Data Model ---

    @dataclass
    class ConversationNote:
        provider: str
        title: str
        date: datetime
        content: str
        images: list[str] = field(default_factory=list)
        tags: list[str] = field(default_factory=list)

    # --- Base Parser ---

    class BaseParser(ABC):
        @abstractmethod
        def can_parse(self, file_path: Path) -> bool:
            """Return True if this parser handles the given file."""

        @abstractmethod
        def parse(self, file_path: Path) -> ConversationNote:
            """Parse file and return a ConversationNote."""

        def clean_content(self, text: str) -> str:
            """Remove HTML tags and normalize whitespace."""
            text = re.sub(r'<[^>]+>', '', text)
            text = re.sub(r'\n{3,}', '\n\n', text)
            return text.strip()

    # --- Provider Parsers ---

    class ClaudeParser(BaseParser):
        def can_parse(self, file_path: Path) -> bool:
            return file_path.suffix == '.json' and 'claude' in file_path.stem.lower()

        def parse(self, file_path: Path) -> ConversationNote:
            import json
            data = json.loads(file_path.read_text())
            # Claude export format: {name, created_at, messages: [{role, content}]}
            title = data.get('name', file_path.stem)
            date = datetime.fromisoformat(data.get('created_at', datetime.now().isoformat()))
            messages = data.get('messages', [])
            content = self._format_messages(messages)
            return ConversationNote(provider='claude', title=title, date=date, content=content)

        def _format_messages(self, messages: list) -> str:
            lines = []
            for msg in messages:
                role = msg.get('role', 'unknown').upper()
                text = self.clean_content(str(msg.get('content', '')))
                lines.append(f"**{role}**: {text}\n")
            return '\n'.join(lines)

    class GeminiParser(BaseParser):
        def can_parse(self, file_path: Path) -> bool:
            return file_path.suffix == '.json' and 'gemini' in file_path.stem.lower()

        def parse(self, file_path: Path) -> ConversationNote:
            import json
            data = json.loads(file_path.read_text())
            title = data.get('title', file_path.stem)
            date = datetime.now()
            content = self.clean_content(str(data))
            return ConversationNote(provider='gemini', title=title, date=date, content=content)

    class ChatGPTParser(BaseParser):
        def can_parse(self, file_path: Path) -> bool:
            return file_path.suffix == '.json' and 'chatgpt' in file_path.stem.lower()

        def parse(self, file_path: Path) -> ConversationNote:
            import json
            data = json.loads(file_path.read_text())
            title = data.get('title', file_path.stem)
            date_ts = data.get('create_time', 0)
            date = datetime.fromtimestamp(date_ts) if date_ts else datetime.now()
            content = self._format_mapping(data.get('mapping', {}))
            return ConversationNote(provider='chatgpt', title=title, date=date, content=content)

        def _format_mapping(self, mapping: dict) -> str:
            messages = []
            for node in mapping.values():
                msg = node.get('message')
                if not msg:
                    continue
                role = msg.get('author', {}).get('role', 'unknown').upper()
                parts = msg.get('content', {}).get('parts', [])
                text = self.clean_content(' '.join(str(p) for p in parts if isinstance(p, str)))
                if text:
                    messages.append(f"**{role}**: {text}\n")
            return '\n'.join(messages)

    class CodexParser(BaseParser):
        def can_parse(self, file_path: Path) -> bool:
            return file_path.suffix == '.json' and 'codex' in file_path.stem.lower()

        def parse(self, file_path: Path) -> ConversationNote:
            import json
            data = json.loads(file_path.read_text())
            title = data.get('title', file_path.stem)
            date = datetime.now()
            content = self.clean_content(str(data))
            return ConversationNote(provider='codex', title=title, date=date, content=content)

    # --- Image Extractor ---

    def extract_images(content: str, source_dir: Path, screenshots_dir: Path) -> tuple[str, list[str]]:
        """Find image references in content, copy to screenshots_dir, return updated content and paths."""
        screenshots_dir.mkdir(parents=True, exist_ok=True)
        img_pattern = re.compile(r'!\[([^\]]*)\]\(([^)]+)\)')
        extracted = []

        def replace_image(match):
            alt, src = match.group(1), match.group(2)
            src_path = source_dir / src if not Path(src).is_absolute() else Path(src)
            if src_path.exists():
                dest = screenshots_dir / src_path.name
                shutil.copy2(src_path, dest)
                extracted.append(str(dest))
                return f'![{alt}]({dest})'
            return match.group(0)

        updated = img_pattern.sub(replace_image, content)
        return updated, extracted

    # --- Note Writer ---

    def write_note(note: ConversationNote, output_dir: Path) -> Path:
        """Write ConversationNote as Obsidian markdown file."""
        output_dir.mkdir(parents=True, exist_ok=True)
        slug = re.sub(r'[^\w\s-]', '', note.title).strip().replace(' ', '-').lower()[:60]
        date_str = note.date.strftime('%Y-%m-%d')
        filename = f"{date_str}-{note.provider}-{slug}.md"
        out_path = output_dir / filename

        frontmatter = f"""---
title: {note.title}
provider: {note.provider}
date: {date_str}
type: conversation
tags:
  - ai-conversation
  - {note.provider}
---

"""
        out_path.write_text(frontmatter + note.content)
        return out_path

    # --- Main ---

    PARSERS = [ClaudeParser(), GeminiParser(), ChatGPTParser(), CodexParser()]

    def sync_file(file_path: Path, output_dir: Path, screenshots_dir: Path) -> None:
        for parser in PARSERS:
            if parser.can_parse(file_path):
                note = parser.parse(file_path)
                note.content, note.images = extract_images(
                    note.content, file_path.parent, screenshots_dir
                )
                out = write_note(note, output_dir)
                print(f"[revolve] Synced: {file_path.name} -> {out}")
                return
        print(f"[revolve] Skipped (no matching parser): {file_path.name}")

    def main():
        parser = argparse.ArgumentParser(description='Revolve conversation sync')
        parser.add_argument('files', nargs='+', help='Files to sync')
        parser.add_argument('--output-dir', required=True, help='Output directory for notes')
        parser.add_argument('--screenshots-dir', required=True, help='Directory for extracted images')
        args = parser.parse_args()

        output_dir = Path(args.output_dir).expanduser()
        screenshots_dir = Path(args.screenshots_dir).expanduser()

        for f in args.files:
            sync_file(Path(f).expanduser(), output_dir, screenshots_dir)

    if __name__ == '__main__':
        main()

    <!DOCTYPE plist PUBLIC "-//Apple//DTD PLIST 1.0//EN"
      "http://www.apple.com/DTDs/PropertyList-1.0.dtd">
    <plist version="1.0">
    <dict>
      <key>Label</key>
      <string>com.revolve.sync</string>

      <key>ProgramArguments</key>
      <array>
        <string>/usr/bin/fswatch</string>
        <string>-0</string>
        <string>-e</string>
        <string>.*</string>
        <string>-i</string>
        <string>\.json$</string>

      <string>/tmp/revolve-sync-error.log</string>

      <key>RunAtLoad</key>
      <false/>

      <key>KeepAlive</key>
      <true/>
    </dict>
    </plist>

    # Scripts

    ## sync_conversations.py

    Syncs AI conversation exports to Obsidian vault.

    ### Usage

    ```bash
    python3 scripts/sync_conversations.py \
      --output-dir "~/Knowledge Base/Research" \
      --screenshots-dir "~/Knowledge Base/Research/screenshots" \
      path/to/conversation.json
    ```

    ### Supported Providers

    | Provider | File Pattern | Export Location |
    |----------|-------------|-----------------|
    | Claude | `claude-*.json` | claude.ai > Settings > Export |
    | Gemini | `gemini-*.json` | myaccount.google.com > Data export |
    | ChatGPT | `chatgpt-*.json` | chat.openai.com > Settings > Export |
    | Codex | `codex-*.json` | platform.openai.com > Export |

    ## com.revolve.sync.plist

    macOS launchd agent for automatic sync via fswatch.

    ### Setup

    1. Run `/revolve-setup` to configure paths
    2. The setup wizard will install and configure this plist automatically
    3. Manual install:
       ```bash
       cp scripts/com.revolve.sync.plist ~/Library/LaunchAgents/
       # Edit WATCH_DIR_PLACEHOLDER with your export directory
       launchctl load ~/Library/LaunchAgents/com.revolve.sync.plist
       ```

    ### View Logs

    ```bash
    tail -f /tmp/revolve-sync.log
    ```
